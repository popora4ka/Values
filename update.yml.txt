name: Auto Update Prices

on:
  schedule:
    - cron: '0 0 * * *'
  workflow_dispatch:

jobs:
  update-prices:
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4
      
      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.10'
          
      - name: Install dependencies
        run: pip install curl_cffi beautifulsoup4 lxml
        
      - name: Run Parser
        run: python auto_parser.py
        
      - name: Commit and Push
        run: |
          git config --global user.name 'GitHub Action Bot'
          git config --global user.email 'action@github.com'
          git add prices.json
          git diff --quiet && git diff --staged --quiet || (git commit -m "Auto-update MM2 prices" && git push)
